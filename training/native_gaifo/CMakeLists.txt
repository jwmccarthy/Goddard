cmake_minimum_required(VERSION 3.18)
project(goddard_gaifo_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenMP REQUIRED)
if(NOT TORCH_ROOT OR NOT UPSTREAM)
  message(FATAL_ERROR "Use gaifo_cpu.py to supply the installed Torch and pinned upstream paths")
endif()
add_compile_definitions(_GLIBCXX_USE_CXX11_ABI=${TORCH_ABI})
set(RS "${UPSTREAM}/RLGymPPO_CPP/RLGymSim_CPP/RocketSim")
add_subdirectory("${RS}" "rocketsim")
add_executable(gaifo_native main.cpp environment.cpp)
target_include_directories(gaifo_native PRIVATE "${RS}/src"
  "${UPSTREAM}/RLGymPPO_CPP/libsrc/json" "${TORCH_ROOT}/include"
  "${TORCH_ROOT}/include/torch/csrc/api/include")
target_link_directories(gaifo_native PRIVATE "${TORCH_ROOT}/lib")
target_link_libraries(gaifo_native PRIVATE RocketSim OpenMP::OpenMP_CXX torch torch_cpu c10)
if(EXISTS "${TORCH_ROOT}/lib/libtorch_cuda.so")
  target_link_libraries(gaifo_native PRIVATE torch_cuda c10_cuda)
endif()
set_target_properties(gaifo_native PROPERTIES BUILD_RPATH "${TORCH_ROOT}/lib")
